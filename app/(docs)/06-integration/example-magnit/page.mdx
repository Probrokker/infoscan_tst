# Пример SOAP/XML интеграции

Многие enterprise-системы (1С через WS-ссылки, Manhattan, SAP, корпоративные шины) используют SOAP с XML-телом. Инфоскан поддерживает это через тот же `http` транспорт — меняется только Content-Type и формат body.

## Подтверждение наименования через SOAP

Сначала Item Type запрашивает наименование товара по ШК. Сервер отвечает SOAP-конвертом.

```javascript
return [
  function (args) {
    return {
      page: 'ConfirmBarcodePage',
      pageTitle: 'Подтвердите товар:',
      nameResult: 'resultConfirm',
      http: {
        url: 'http://corporate-wms.example.ru/soap/catalog',
        headers: {
          'Accept': 'application/xml',
          'Content-Type': 'application/xml',
          'SOAPAction': 'CheckBarcode',
          'Authorization': 'Basic ' + HELPER_JS.toBase64('infoscan:Sg5Tk8Vp9Qx2')
        },
        body: function () {
          return '<soap:Envelope xmlns:soap="http://schemas.xmlsoap.org/soap/envelope/">' +
                 '<soap:Body>' +
                 '<m:CheckBarcode xmlns:m="https://iscan.example.ru">' +
                 '<m:Barcode>' + BARCODE + '</m:Barcode>' +
                 '</m:CheckBarcode>' +
                 '</soap:Body>' +
                 '</soap:Envelope>'
        },
        resp: function (resp) {
          var json = JSON.parse(HELPER_JS.xmlToJson(resp))
          var status = json.Envelope.Body.CheckBarcodeResponse.Status['#text']

          if (status !== 'OK') {
            var comment = json.Envelope.Body.CheckBarcodeResponse.Comment['#text']
            throw Error(comment || 'Товар не найден')
          }

          return json.Envelope.Body.CheckBarcodeResponse.Comment['#text']
        }
      }
    }
  }
]
```

## Отправка измерения

`jsScriptFinishSend` со SOAP:

```javascript
var len = SENSOR.lengthFormat || SENSOR.depthFormat || 0

return {
  http: function () {
    return {
      url: 'http://corporate-wms.example.ru/soap/measurements',
      headers: {
        'Accept': 'application/xml',
        'Content-Type': 'application/xml',
        'SOAPAction': 'LoadMeasurement',
        'Authorization': 'Basic ' + HELPER_JS.toBase64('infoscan:Sg5Tk8Vp9Qx2')
      },
      body: function () {
        return '<soap:Envelope xmlns:soap="http://schemas.xmlsoap.org/soap/envelope/">' +
               '<soap:Body>' +
               '<LoadMeasurement xmlns="https://iscan.example.ru">' +
               '<Barcode>' + BARCODE + '</Barcode>' +
               '<Width>' + SENSOR.widthFormat + '</Width>' +
               '<Height>' + SENSOR.heightFormat + '</Height>' +
               '<Length>' + len + '</Length>' +
               '<Weight>' + SENSOR.weightFormat + '</Weight>' +
               '<DeviceCode>' + DEV_CODE + '</DeviceCode>' +
               '<MeasuredAt>' + new Date().toISOString() + '</MeasuredAt>' +
               '</LoadMeasurement>' +
               '</soap:Body>' +
               '</soap:Envelope>'
      },
      resp: function (resp, httpCode) {
        if (httpCode !== 200) {
          throw Error('Сервер вернул HTTP ' + httpCode)
        }

        // Альтернативный путь: XPath напрямую (быстрее, чем xmlToJson)
        var code = HELPER_JS.xmlXPath(resp, '//Status/text()')

        if (code === '0' || code === 'OK') {
          return {
            banner: BARCODE + ' — записан',
            bannerColor: '#22C55E',
            comment: ''
          }
        }

        var message = HELPER_JS.xmlXPath(resp, '//Comment/text()')
        throw Error(message || ('Сервер вернул код: ' + code))
      }
    }
  }
}
```

## Заголовки X-Tander-style

Если корпоративная WMS использует кастомные заголовки трассировки — добавьте их в `headers`:

```javascript
headers: {
  'Content-Type': 'application/xml',
  'SOAPAction': 'LoadMeasurement',
  'Authorization': 'Basic ' + HELPER_JS.toBase64('infoscan:password'),
  'X-Tander-Code-DC': '042',                       // код распределительного центра
  'X-Tander-Barcode': BARCODE,                     // дублирование ШК в заголовке
  'X-Tander-Request-Id': String(Math.floor(Math.random() * 1000000000))
}
```

## Парсинг SOAP-фолтов

Если сервер вернул `soap:Fault` (стандартный конверт ошибки SOAP), HTTP-код будет 500. Парсить тело можно так:

```javascript
resp: function (resp, httpCode) {
  if (httpCode === 500) {
    var fault = HELPER_JS.xmlXPath(resp, '//faultstring/text()')
    throw Error('SOAP Fault: ' + (fault || 'неизвестная ошибка'))
  }
  if (httpCode !== 200) {
    throw Error('HTTP ' + httpCode)
  }
  // ... обычная обработка
}
```

## Где смотреть XML «сырым»

Если ответ парсится странно — посмотрите его «сырым» через временный шаблон-дебаг:

```javascript
resp: function (resp, httpCode) {
  console.log('HTTP', httpCode, 'BODY:', resp)   // в журнал устройства
  throw Error('DEBUG: см. журнал')
}
```

Журналы устройства — Личный кабинет → системные логи (или через `tail` по SSH, если у вашей версии PRO открыт SSH).

## SOAP без WSDL — нормально?

Да. Инфоскан **не парсит WSDL** и не валидирует пространства имён. Главное — чтобы XML был корректным синтаксически и сервер его принял. Если ваша корпоративная политика требует строгого соответствия WSDL — поставьте перед Инфосканом прокси, который дополнит конверт.
